\chapter{Mạng nơ-ron tích chập (CNN)}\label{chap:cnn}

Ảnh $224\times224\times3$ có hơn 150 nghìn điểm giá trị; nối đầy đủ vào một tầng 1000 nơ-ron đã cần hơn 150 triệu tham số và làm mất cấu trúc không gian. CNN giải quyết việc này bằng hai ý tưởng lấy cảm hứng từ
vỏ não thị giác \cite{ch3:hubel1962,ch3:fukushima1980}: \emph{kết nối cục bộ} và \emph{chia sẻ trọng số} \cite{ch3:lecun1989,ch1:lecun1998}. Chương này trình bày cơ sở toán học, tiến hoá kiến trúc, thực nghiệm trên bốn bộ ảnh với
ba nền tảng (NumPy, PyTorch, TensorFlow--Keras), rồi triển khai lên Web App.

\section{Cơ sở lý thuyết}\label{sec:cnn-theory}
\hinhtikz{ch3_cnn_general}{Cấu trúc tổng quát của một CNN phân loại ảnh màu: hai khối Conv -- ReLU -- Pool trích đặc trưng, tiếp theo là các tầng kết nối đầy đủ}{ch3-general}

\subsection{Phép tích chập hai chiều}
Với ảnh vào $I$ và kernel $K$ kích thước $k\times k$, bản đồ đặc trưng tại vị trí $(i,j)$ là
\begin{equation}
  S(i,j)=\sum_{u=0}^{k-1}\sum_{v=0}^{k-1} I(i+u,j+v)\,K(u,v).
\end{equation}
Với kích thước vào $n$, đệm $p$, bước $s$, kích thước ra và số tham số của một tầng có $c_{\text{in}}$ kênh vào, $c_{\text{out}}$ kênh ra là
\begin{equation}
  n'=\left\lfloor\frac{n+2p-k}{s}\right\rfloor+1,\qquad \#\text{tham số}=(k^2c_{\text{in}}+1)\,c_{\text{out}}.
\end{equation}
Số tham số không phụ thuộc kích thước ảnh, đây là lợi ích của chia sẻ trọng số. Hình~\ref{fig:ch3-conv} minh hoạ một kernel phát hiện cạnh dọc.

\hinhtikz{ch3_conv_op}{Phép tích chập trên ảnh $6\times6$ với kernel phát hiện cạnh dọc $3\times3$, bước 1, không đệm}{ch3-conv}

\subsection{Pooling, vùng tiếp nhận và hàm kích hoạt}
Pooling giảm kích thước không gian và tạo tính bất biến nhỏ với phép dịch (Hình~\ref{fig:ch3-pooling}). \emph{Vùng tiếp nhận} (receptive field) là phần ảnh gốc mà một nơ-ron ``nhìn thấy''; qua các tầng nó tăng theo
\begin{equation}
  r_l=r_{l-1}+(k_l-1)\prod_{i=1}^{l-1}s_i,\qquad r_0=1.
\end{equation}
Hai tầng $3\times3$ có vùng tiếp nhận $5\times5$ với ít tham số hơn một tầng $5\times5$, cơ sở thiết kế của VGG. ReLU $\max(0,x)$ \cite{ch3:nair2010} tránh bão hoà gradient; tầng cuối dùng Softmax kết hợp entropy chéo.
Batch Normalization \cite{ch2:ioffe2015} chuẩn hoá theo lô, Dropout \cite{ch2:srivastava2014} tắt ngẫu nhiên nơ-ron, còn tăng cường dữ liệu \cite{ch3:shorten2019} tạo biến thể ảnh để giảm quá khớp.

\hinhtikz{ch3_pooling}{Max pooling và average pooling $2\times2$, bước 2 trên bản đồ $4\times4$}{ch3-pooling}

\subsection{Tiến hoá kiến trúc}
LeNet-5 \cite{ch1:lecun1998} đặt khuôn mẫu Conv -- Pool -- FC. AlexNet \cite{ch1:krizhevsky2012} chứng minh sức mạnh của mạng sâu với ReLU, Dropout và GPU trên ImageNet \cite{ch3:russakovsky2015}. VGG \cite{ch3:simonyan2015} chỉ dùng kernel $3\times3$
xếp chồng (Hình~\ref{fig:ch3-vgg}); GoogLeNet \cite{ch3:szegedy2015} tính song song nhiều cỡ kernel.

\hinhtikz{ch3_lenet5}{Kiến trúc LeNet-5: C1, C3 là tích chập $5\times5$; S2, S4 là giảm mẫu; F5, F6 và tầng ra kết nối đầy đủ}{ch3-lenet}
\hinhtikz{ch3_vgg16}{Kiến trúc VGG-16: năm khối gồm 2--3 tầng Conv $3\times3$, sau mỗi khối là Max Pool, cuối cùng là ba tầng FC}{ch3-vgg}

Khi mạng quá sâu, sai số huấn luyện lại tăng vì khó tối ưu. ResNet \cite{ch3:he2016} cho mỗi khối học phần dư $F(\mathbf{x})$ thay vì ánh xạ đầy đủ:
\begin{equation}
  \mathbf{y}=F(\mathbf{x},\{W_i\})+\mathbf{x}.
\end{equation}
Lối tắt cho gradient đi thẳng về các tầng đầu (Hình~\ref{fig:ch3-res}). DenseNet \cite{ch3:huang2017} nối mọi tầng với nhau. MobileNet \cite{ch3:howard2017,ch3:sandler2018} tách tích chập thành depthwise và point-wise, giảm chi phí theo tỷ lệ
\begin{equation}
  \frac{D_k^2MD_F^2+MND_F^2}{D_k^2MND_F^2}=\frac1N+\frac1{D_k^2}.
\end{equation}
EfficientNet \cite{ch3:tan2019} co giãn đồng thời độ sâu, độ rộng và độ phân giải. Với kernel $3\times3$ và 256 kênh, MobileNet chỉ tốn khoảng 11,5\% phép nhân so với tích chập chuẩn (Hình~\ref{fig:ch3-depthwise}).

\hinhtikz{ch3_residual}{Khối dư: nhánh $F(\mathbf{x})$ cộng với lối tắt, cho gradient đi thẳng về các tầng đầu}{ch3-res}
\hinhtikz{ch3_depthwise}{(a) Tích chập chuẩn; (b) tích chập tách được theo chiều sâu của MobileNet}{ch3-depthwise}

\section{Khảo sát dữ liệu}\label{sec:cnn-data}
Mọi mô hình đọc cùng một tệp dữ liệu và cùng một phép chia. 10\% tập huấn luyện gốc được tách thành Validation theo từng lớp; tập kiểm tra gốc chỉ dùng một lần ở bước cuối (Bảng~\ref{tab:ch3-data}).

\bang{ch3_data_summary}{Bốn bộ dữ liệu ảnh của chương sau khi tải và chia tập}{ch3-data}

\begin{itemize}
  \item \textbf{MNIST} \cite{ch1:lecun1998}: chữ số viết tay, ảnh xám, nền đen sạch, các lớp cân bằng; là bài toán dễ dùng để kiểm chứng cài đặt.
  \item \textbf{Fashion-MNIST} \cite{ch3:xiao2017}: cùng định dạng nhưng là trang phục; các lớp áo sơ mi, áo phông, áo khoác dễ nhầm với nhau.
  \item \textbf{Rock/Paper/Scissors} \cite{ch3:moroney2019}: ảnh bàn tay dựng bằng đồ hoạ máy tính, tập kiểm tra có kiểu bàn tay không xuất hiện khi huấn luyện, nên độ dịch chuyển phân phối lớn.
  \item \textbf{CIFAR-10} \cite{ch1:krizhevsky2009}: ảnh màu $32\times32$ của vật thể thật, nền phức tạp, độ phân giải thấp, là bộ khó nhất.
\end{itemize}

\subsection{Diễn giải mô hình bằng Grad-CAM}
Grad-CAM \cite{ch3:selvaraju2017} tính bản đồ nhiệt từ gradient của điểm lớp theo bản đồ đặc trưng $A^k$ tầng tích chập cuối:
\begin{equation}
  \alpha_k^c=\frac1Z\sum_{i,j}\frac{\partial y^c}{\partial A^k_{ij}},\qquad L^c=\mathrm{ReLU}\Big(\sum_k\alpha_k^cA^k\Big).
\end{equation}

\nhanxet{Với ``bao'' và ``kéo'', vùng nóng nằm đúng ở các ngón tay; cổ tay và nền không đóng góp. Với ``búa'', vùng ảnh hưởng lan lên khoảng trống phía trên nắm tay, hợp lý vì điều phân biệt lớp này chính là việc không có ngón tay ở đó. Trên CIFAR-10, mô hình còn dựa vào cả nền, một rủi ro cần lưu ý khi suy rộng.}

\section{Tổng kết chương}
Bốn kết luận chính được rút ra. Thứ nhất, framework không làm mô hình thông minh hơn: Basic CNN viết bằng NumPy đạt độ chính xác ngang bản PyTorch và TensorFlow cùng kiến trúc trên MNIST, framework chỉ làm mô hình nhanh hơn và dễ mở rộng hơn. Thứ hai, kiến trúc quan trọng hơn khi bài toán khó:
khối dư, BatchNorm và tăng cường dữ liệu mang lại \SL{ch3_cifar10/gain_torch/accPct} điểm trên CIFAR-10 (đạt \SL{ch3_cifar10/torch_improved/accPct}\%). Thứ ba, dữ liệu quyết định giới hạn: lỗi trên Fashion-MNIST là các cặp áo có nhãn mơ hồ, lỗi trên Rock/Paper/Scissors chủ yếu là kiểu bàn tay chưa
từng thấy khi huấn luyện. Thứ tư, CNN giải thích được một phần bằng Grad-CAM. CNN xử lý tốt dữ liệu không gian nhưng không có trí nhớ theo thời gian; chương sau trình bày mạng hồi quy cho dữ liệu chuỗi.

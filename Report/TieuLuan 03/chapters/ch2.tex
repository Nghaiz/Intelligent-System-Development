\chapter{Các kỹ thuật học máy cơ bản}\label{chap:ml}

Theo Mitchell \cite{ch2:mitchell1997}, một chương trình \emph{học} từ kinh nghiệm $E$ đối với nhiệm vụ $T$ và độ đo $P$ nếu hiệu năng
trên $T$, đo bằng $P$, cải thiện nhờ $E$. Chương này trình bày các thuật toán cốt lõi, khảo sát ba bộ dữ liệu (tiểu đường, giá nhà, nhận
xét khách hàng), cài đặt và đối chuẩn mô hình, mở rộng sang MLP trên ba nền tảng, rồi đóng gói thành dịch vụ web.

\section{Cơ sở lý thuyết}\label{sec:ml-theory}
\subsection{Bài toán học có giám sát và quy trình}
Tập dữ liệu gồm $n$ cặp $\mathcal{D}=\{(\mathbf{x}_i,y_i)\}_{i=1}^n$ với $\mathbf{x}_i\in\mathbb{R}^d$. Nhãn $y_i$ rời rạc ứng với phân
loại, liên tục ứng với hồi quy. Huấn luyện là cực tiểu hoá rủi ro thực nghiệm, thường có thêm hạng chính quy hoá:
\begin{equation}
  \hat{\boldsymbol\theta}=\arg\min_{\boldsymbol\theta}\frac1n\sum_{i=1}^n\ell\big(f_{\boldsymbol\theta}(\mathbf{x}_i),y_i\big)+\lambda R(\boldsymbol\theta).
\end{equation}
Khi không có nghiệm đóng, tham số được cập nhật theo Gradient Descent:
$\boldsymbol\theta\leftarrow\boldsymbol\theta-\eta\nabla_{\boldsymbol\theta}J$. Quy trình đầy đủ ở Hình~\ref{fig:ch2-pipeline}; dữ liệu được chia
70/15/15 thành Train, Validation, Test, trong đó Test chỉ dùng một lần sau khi đã cố định mô hình.

\hinhtikz{ch2_pipeline}{Quy trình học máy có giám sát tám bước; vòng đứt nét là vòng lặp điều chỉnh}{ch2-pipeline}

\subsection{Hồi quy tuyến tính, Ridge và Lasso}
Mô hình $\hat y=\mathbf{w}^\top\mathbf{x}+b$ cực tiểu sai số bình phương trung bình. Với ma trận thiết kế $X$ (đã thêm cột 1), nghiệm đóng là
phương trình chuẩn:
\begin{equation}
  J(\mathbf{w})=\frac1n\|X\mathbf{w}-\mathbf{y}\|_2^2,\qquad \mathbf{w}^*=(X^\top X)^{-1}X^\top\mathbf{y}.
\end{equation}
Ridge \cite{ch2:hoerl1970} thêm phạt $\lambda\|\mathbf{w}\|_2^2$, co đều các hệ số và cho nghiệm $(X^\top X+\lambda I)^{-1}X^\top\mathbf{y}$. Lasso
\cite{ch2:tibshirani1996} thêm phạt $\lambda\|\mathbf{w}\|_1$, đưa một phần hệ số về đúng 0 nên có tác dụng chọn đặc trưng.

\hinhtikz{ch2_linreg}{Hồi quy tuyến tính một biến: đường $\hat y=w_0+w_1x$ cực tiểu tổng bình phương phần dư}{ch2-linreg}
\maNguon{ch2_scratch__linreg_gd}{Hồi quy tuyến tính bằng Gradient Descent trên MSE (NumPy)}

\subsection{Hồi quy Logistic và Softmax}
Hồi quy logistic \cite{ch2:cox1958} đưa tổ hợp tuyến tính qua hàm sigmoid và ước lượng bằng hợp lý cực đại, tương đương cực tiểu entropy chéo:
\begin{equation}
  p=\sigma(\mathbf{w}^\top\mathbf{x}+b)=\frac{1}{1+e^{-(\mathbf{w}^\top\mathbf{x}+b)}},\quad
  J=-\frac1n\sum_{i=1}^n\big[y_i\log p_i+(1-y_i)\log(1-p_i)\big].
\end{equation}
Với $K$ lớp, Softmax Regression thay sigmoid bằng
$P(y=k\mid\mathbf{x})=\exp(\mathbf{w}_k^\top\mathbf{x})/\sum_{j}\exp(\mathbf{w}_j^\top\mathbf{x})$. Nhãn được quyết định bằng cách so xác suất với một
ngưỡng; ngưỡng 0,5 không nhất thiết tốt nhất khi lớp mất cân bằng.

\subsection{Cây quyết định và phương pháp tổ hợp}
Cây quyết định \cite{ch2:breiman1984,ch2:quinlan1986} chia không gian đặc trưng theo các ngưỡng, ở mỗi nút chọn phép tách giảm độ vẩn đục nhiều nhất.
Với $p_k$ là tỷ lệ lớp $k$ trong nút, chỉ số Gini và entropy là
\begin{equation}
  G=1-\sum_{k}p_k^2,\qquad H=-\sum_{k}p_k\log_2 p_k.
\end{equation}
Cây dễ diễn giải nhưng phương sai cao. Hai hướng tổ hợp khắc phục điểm này (Hình~\ref{fig:ch2-ensemble}): \textbf{Bagging} và Random Forest
\cite{ch2:breiman1996,ch2:breiman2001} huấn luyện các cây độc lập trên mẫu bootstrap rồi bỏ phiếu, giảm phương sai;
\textbf{Boosting} \cite{ch2:friedman2001} xây cây tuần tự, cây sau khớp sai số của các cây trước, giảm thiên lệch. XGBoost \cite{ch2:chen2016} và
LightGBM \cite{ch2:ke2017} là hai cài đặt gradient boosting hiệu quả.

\hinhtikz{ch2_tree}{Cây quyết định độ sâu 2 huấn luyện trên hồ sơ tiểu đường; ngưỡng và tỷ lệ được đọc từ mô hình thật}{ch2-tree}
\hinhtikz{ch2_ensemble}{Bagging (trái): các cây độc lập trên mẫu bootstrap, kết quả bỏ phiếu. Boosting (phải): mỗi cây sửa sai cho các cây trước}{ch2-ensemble}

\subsection{Máy vector hỗ trợ và các thuật toán khác}
SVM \cite{ch2:cortes1995} tìm siêu phẳng có lề lớn nhất, cho phép vi phạm qua biến nới lỏng $\xi_i$:
\begin{equation}
  \min_{\mathbf{w},b,\boldsymbol\xi}\ \frac12\|\mathbf{w}\|^2+C\sum_{i}\xi_i\quad\text{với}\quad y_i(\mathbf{w}^\top\mathbf{x}_i+b)\ge1-\xi_i,\ \ \xi_i\ge0.
\end{equation}
Thủ thuật kernel thay tích vô hướng bằng $k(\mathbf{x},\mathbf{x}')=\exp(-\gamma\|\mathbf{x}-\mathbf{x}'\|^2)$ để có biên phi tuyến; SVR \cite{ch2:drucker1997} áp dụng ý
tưởng này cho hồi quy. KNN \cite{ch2:cover1967} dự đoán theo $k$ láng giềng gần nhất; Naive Bayes \cite{ch2:mccallum1998} áp dụng định lý Bayes với giả
định các đặc trưng độc lập có điều kiện theo lớp.

\hinhtikz{ch2_svm_kernel}{Kernel trick: hai vòng tròn đồng tâm không tách được trong $\mathbb{R}^2$ trở nên tách được khi ánh xạ lên chiều cao hơn}{ch2-kernel}

\subsection{Đánh giá mô hình và quá khớp}
Từ ma trận nhầm lẫn (Hình~\ref{fig:ch2-confusion}) ta có Precision $=\mathrm{TP}/(\mathrm{TP}+\mathrm{FP})$, Recall $=\mathrm{TP}/(\mathrm{TP}+\mathrm{FN})$
và F1 là trung bình điều hoà của hai đại lượng này. ROC-AUC \cite{ch2:fawcett2006} đo khả năng xếp hạng; với dữ liệu mất cân bằng, đường
Precision--Recall có tính thông tin cao hơn \cite{ch2:saito2015}. Cho hồi quy, ta dùng MAE, RMSE và hệ số xác định $R^2=1-\sum_i(y_i-\hat y_i)^2/\sum_i(y_i-\bar y)^2$.

\hinhtikz{ch2_confusion}{Ma trận nhầm lẫn và bốn chỉ số phân loại suy ra từ nó}{ch2-confusion}

Sai số kỳ vọng phân rã thành thiên lệch bình phương, phương sai và nhiễu không giảm được \cite{ch2:geman1992}. Hình~\ref{fig:ch2-bv} cho thấy
sai số Test có dạng chữ U theo độ phức tạp: mô hình quá đơn giản thì chưa khớp, quá phức tạp thì quá khớp.

\hinhpdf{ch2_bias_variance}{0.92}{Sai số Train giảm đơn điệu theo bậc đa thức, còn sai số Test có dạng chữ U}{ch2-bv}

\section{Mạng nơ-ron đa tầng trên ba nền tảng}\label{sec:ml-mlp}
\subsection{Kiến trúc và lan truyền ngược}
MLP xếp chồng các tầng $\mathbf{h}^{(l)}=\phi(W^{(l)}\mathbf{h}^{(l-1)}+\mathbf{b}^{(l)})$ với hàm kích hoạt phi tuyến ReLU. Lan truyền ngược \cite{ch1:rumelhart1986} áp dụng quy tắc chuỗi từ đầu ra về đầu vào:
\begin{equation}
  \frac{\partial J}{\partial W^{(l)}}=\boldsymbol\delta^{(l)}\,\mathbf{h}^{(l-1)\top},\qquad
  \boldsymbol\delta^{(l-1)}=\phi'\big(\mathbf{z}^{(l-1)}\big)\odot W^{(l)\top}\boldsymbol\delta^{(l)},
\end{equation}
với $\boldsymbol\delta^{(l)}=\partial J/\partial\mathbf{z}^{(l)}$ và $\mathbf{z}^{(l)}$ là đầu vào của hàm kích hoạt tầng $l$. Tối ưu dùng Adam \cite{ch2:kingma2015}; BatchNorm \cite{ch2:ioffe2015} và Dropout \cite{ch2:srivastava2014} ổn định huấn luyện và giảm quá khớp.
Cùng kiến trúc (Hình~\ref{fig:ch2-mlp}) được cài đặt trên ba nền tảng để đối chiếu.

\hinhtikz{ch2_mlp}{MLP cho dữ liệu bảng: ba lớp ẩn 128--64--32 (Linear, BatchNorm, ReLU, Dropout) và một đầu ra}{ch2-mlp}

\subsection{Hiện thực bằng NumPy}
Bản từ đầu tự viết lan truyền thuận, lan truyền ngược và Adam, không có BatchNorm và Dropout. Gradient được kiểm tra bằng sai phân số trước khi huấn luyện, sai số tương đối dưới $10^{-9}$.

chọn mặc định cho dữ liệu bảng; khi cùng kiến trúc và dữ liệu, nền tảng ảnh hưởng tới kết quả ít hơn so với việc chọn đúng họ mô hình.}

\section{Tổng kết chương}
Ba bài học được rút ra. Thứ nhất, chỉ số phải khớp với bài toán: accuracy cao có thể vô nghĩa trên dữ liệu mất cân bằng. Thứ hai, không có mô hình tốt nhất cho mọi dữ liệu: gradient boosting thắng khi quan hệ có dạng ngưỡng,
Lasso thắng khi quan hệ gần tuyến tính, Logistic Regression trên TF-IDF thắng với văn bản ngắn. Thứ ba, quy trình quan trọng không kém thuật toán: làm sạch có hiểu biết miền, tránh rò rỉ, chọn ngưỡng trên Validation.

\chapter{Hệ đa tác tử (Multi-Agent Systems)}\label{chap:mas}

\section{Tổng quan và cơ sở lý thuyết}\label{sec:mas-theory}
\subsection{Từ tác tử đơn sang đa tác tử}
Tác tử là thực thể \emph{cảm nhận} môi trường và \emph{tác động} lên môi trường \cite{ch7:russell2021}, có tính tự chủ, phản ứng, chủ động và khả năng xã hội \cite{ch7:wooldridge2009}. Trong hệ hiện đại, bộ não của tác tử thường là LLM, vận hành theo vòng lặp quan sát -- suy luận -- hành động -- ghi nhớ \cite{ch1:yao2023} (Hình~\ref{fig:mas-loop}). Hệ đa tác tử chia nhiệm vụ cho nhiều vai trò chuyên biệt, nhằm tách biệt ngữ cảnh, kiểm chứng chéo và song song hoá.

\hinhtikz{ch7_agent_loop}{Vòng lặp của một tác tử dựa trên LLM: quan sát, suy luận, hành động trên môi trường}{mas-loop}

\subsection{Cơ sở hình thức}
Khi các tác tử cùng mục tiêu nhưng chỉ quan sát được một phần môi trường, bài toán được mô hình hoá bằng Dec-POMDP, bộ tám phần tử
\begin{equation}
  \mathcal{M}=\big\langle \mathcal{I},\mathcal{S},\{\mathcal{A}_i\},T,R,\{\Omega_i\},O,\gamma\big\rangle,
\end{equation}
gồm tập tác tử $\mathcal{I}$, không gian trạng thái $\mathcal{S}$, tập hành động $\mathcal{A}_i$, hàm chuyển $T(s'\mid s,\mathbf{a})$, phần thưởng chung $R(s,\mathbf{a})$, tập quan sát $\Omega_i$, hàm quan sát $O$ và hệ số chiết khấu $\gamma$. Mục tiêu là tìm chính sách chung $\boldsymbol\pi=(\pi_1,\dots,\pi_n)$ cực đại
\begin{equation}
  J(\boldsymbol\pi)=\mathbb{E}\Big[\sum_{t=0}^{\infty}\gamma^t R(s_t,\mathbf{a}_t)\Big].
\end{equation}
Nếu mỗi tác tử có phần thưởng riêng, mô hình trở thành trò chơi Markov. Ở mức nhận thức, mô hình BDI mô tả tác tử bằng niềm tin (Belief), mong muốn (Desire) và ý định (Intention) \cite{ch7:wooldridge2009}: trong hệ LLM, ba thành phần này tương ứng trạng thái bộ nhớ, mục tiêu và kế hoạch đã cam kết.

\section{Kiến trúc phối hợp và giao thức}\label{sec:mas-arch}
\subsection{Các cấu trúc phối hợp}
Cách nối các tác tử quyết định luồng thông tin, điểm nghẽn và chế độ hỏng (Hình~\ref{fig:mas-patterns}). Mô hình \textbf{điều phối -- thợ} có một Supervisor phân việc và tổng hợp; \textbf{đường ống} truyền kết quả tuần tự; \textbf{tranh luận} cho các tác tử phản biện nhau \cite{ch7:du2024debate}; \textbf{bảng đen} dùng vùng nhớ chung \cite{ch7:erman1980}.

Hai giao thức bổ sung cho nhau (Hình~\ref{fig:mas-protocols}): MCP nối tác tử với công cụ và dữ liệu \cite{ch7:mcp2024}, A2A nối tác tử với tác tử \cite{ch7:a2a2025}. Thông điệp có cấu trúc chứa người gửi, người nhận, loại, nội dung và định danh tác vụ, giúp truy vết toàn bộ hội thoại.

\section{Thực nghiệm: đa tác tử giải toán lời văn GSM8K}\label{sec:mas-exp}
\subsection{Bài toán và câu hỏi nghiên cứu}
GSM8K \cite{ch1:cobbe2021} gồm các bài toán lời văn bậc tiểu học, mỗi bài cần 2--8 phép tính, đáp án là một số; chỉ một bước sai là sai cả câu. Mọi hệ dùng chung LLM Qwen2.5-1.5B-Instruct \cite{ch5:qwen2024} chạy cục bộ trên GPU, giải mã tham lam. Chọn mô hình nhỏ là có chủ ý: tác dụng hay tác hại của việc thêm tác tử sẽ hiện ra rõ. Thực nghiệm trả lời ba câu hỏi: suy luận từng bước và công cụ cải thiện bao nhiêu; hệ đa tác tử có tốt hơn tác tử đơn tốt nhất không; phần cải thiện phải trả bằng bao nhiêu chi phí.

\subsection{Thiết kế bốn hệ}
Bốn hệ được so sánh trên cùng tập câu và cùng cách chấm (Hình~\ref{fig:mas-systems}):
\begin{itemize}
  \item \textbf{Trực tiếp:} một lượt gọi, chỉ in ra con số; là mốc dưới.
  \item \textbf{CoT:} một tác tử suy luận từng bước \cite{ch1:wei2022}.
  \item \textbf{PAL + Python:} một tác tử viết hàm \texttt{solution()}, công cụ chạy mã trong hộp cát \cite{ch7:gao2023pal}; nếu lỗi thì gửi lại để sửa một lần.
  \item \textbf{Đa tác tử:} Planner lập kế hoạch; Solver-PAL viết mã, Solver-CoT giải bằng lời độc lập. Nếu hai đáp án khớp, Verifier chấp nhận (cổng đồng thuận). Nếu không, Verifier phân xử, một Solver-PAL thứ hai lấy mẫu ở nhiệt độ 0,7, rồi Coordinator biểu quyết đa số, ý tưởng gần với self-consistency \cite{ch7:wang2023sc}.
\end{itemize}

\section{Tổng kết chương}
Trên \SL{ch7_gsm8k/n} câu chung, suy luận từng bước là đòn bẩy lớn nhất; công cụ không tự động tốt hơn; hệ đa tác tử không vượt tác tử đơn tốt nhất với mô hình nhỏ, nhưng sự đồng thuận giữa các Solver là tín hiệu tin cậy có giá trị. Bài học thiết kế: bắt đầu từ tác tử đơn mạnh nhất; chỉ thêm tác tử khi có tín hiệu kiểm chứng bên ngoài đáng tin như trình thông dịch, bộ kiểm thử hay người duyệt; và dùng sự bất đồng để chuyển câu khó cho người kiểm tra. Giới hạn của chương là cỡ mẫu nhỏ và chỉ thử một LLM.

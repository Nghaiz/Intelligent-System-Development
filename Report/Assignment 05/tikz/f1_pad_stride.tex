% Padding P = 1 và stride S = 2 trên ảnh 5x5, bộ lọc 3x3.
\begin{tikzpicture}[x=5.5mm, y=5.5mm]
  \foreach \i in {-1,...,5} \foreach \j in {-1,...,5} {
    \pgfmathsetmacro{\pad}{(\i<0 || \i>4 || \j<0 || \j>4) ? 1 : 0}
    \ifdim\pad pt>0pt
      \node[o, minimum size=5.5mm, fill=lightGray, text=gray] at (\j, -\i) {0};
    \else
      \node[o, minimum size=5.5mm] at (\j, -\i) {};
    \fi}
  \draw[mOne, very thick] (-1.5, 1.5) rectangle (1.5, -1.5);
  \draw[mTwo, very thick, dashed] (0.5, 1.5) rectangle (3.5, -1.5);
  \draw[mThree, very thick, dotted] (-1.5, -0.5) rectangle (1.5, -3.5);
  \node[above, font=\footnotesize] at (2, 1.6) {ảnh $5\times5$, đệm $P=1$ (ô xám), bước $S=2$};
  % Đầu ra
  \foreach \i in {0,...,2} \foreach \j in {0,...,2}
    \node[o, minimum size=5.5mm] at (9+\j, -1-\i) {};
  \node[o, minimum size=5.5mm, fill=mOne!30] at (9, -1) {};
  \node[o, minimum size=5.5mm, fill=mTwo!30] at (10, -1) {};
  \node[o, minimum size=5.5mm, fill=mThree!30] at (9, -2) {};
  \node[above, font=\footnotesize] at (10, -0.3) {đầu ra $3\times3$};
  \draw[-{Stealth}, thick, gray] (6, -2) -- (8, -2);
  \node[font=\small, anchor=west] at (-1.5, -7.3)
    {$H_{\text{ra}} = \left\lfloor \dfrac{H + 2P - K}{S} \right\rfloor + 1

\end{tikzpicture}

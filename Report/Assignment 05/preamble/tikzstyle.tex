\usepackage{tikz}
\usepackage{pgfplots}
\usepackage{pgfplotstable}
\pgfplotsset{compat=1.18}

                backgrounds,decorations.pathreplacing}

% Externalize: mỗi hình biên dịch một lần thành tikzcache/<tên>.pdf.
\tikzexternalize[prefix=tikzcache/]
% Khi chạy bằng LuaLaTeX, external tự gọi lualatex cho từng hình, không cần khai system call.
% \hinh{tên}: nạp tikz/<tên>.tex, đặt tên tệp cache theo chính tên hình.
\newcommand{\hinh}[1]{\tikzsetnextfilename{#1}\input{tikz/#1}}

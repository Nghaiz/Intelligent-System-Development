% RNN dạng cuộn (trái) và mở theo thời gian (phải): cùng W_xh, W_hh, b ở mọi bước.
\begin{tikzpicture}[node distance=10mm, font=\small, scale=0.88, transform shape,
  nh/.style={hop=cBlue, minimum width=12mm, minimum height=10mm, font=\normalsize},
  nx/.style={hop=cGreen, minimum width=12mm, font=\normalsize},
  ny/.style={hop=cOrange, minimum width=12mm, font=\normalsize}]
  \node[nh] (hc) {$h$};
  \node[nx, below=of hc] (xc) {$x_t$};
  \node[ny, above=of hc] (yc) {$\hat y$};
  \draw[muiM=cGreen!60!black] (xc) -- node[right, font=\small] {$W_{xh}$} (hc);
  \draw[muiM=cOrange!70!black] (hc) -- node[right, font=\small] {$W_{hy}$} (yc);
  \draw[muiM=cRed, line width=1.8pt] (hc.east) .. controls +(1.4,0.9) and +(1.4,-0.9) .. node[right, font=\small, text=cRed] {$W_{hh}$} (hc.east);
  \node[font=\huge, text=cNavy] at (3.0,0) {$\Leftrightarrow$};
  \begin{scope}[xshift=4.2cm]
    \node[hop=cSlate, minimum width=10mm, minimum height=10mm] (h0) {$h_0$};
    \foreach \t/\p in {1/h0, 2/h1, 3/h2} {
      \node[nh, right=13mm of \p] (h\t) {$h_{\t}$};
      \node[nx, below=of h\t] (x\t) {$x_{\t}$};
      \draw[muiM=cGreen!60!black] (x\t) -- node[right, font=\small] {$W_{xh}$} (h\t);
      \draw[muiM=cRed, line width=1.8pt] (\p) -- node[above, font=\small, text=cRed] {$W_{hh}$} (h\t);
    }
    \node[right=7mm of h3, font=\Large] (dots) {$\cdots$};
    \node[nh, right=7mm of dots] (hT) {$h_T$};
    \node[nx, below=of hT] (xT) {$x_T$};
    \draw[muiM=cGreen!60!black] (xT) -- node[right, font=\small] {$W_{xh}$} (hT);
    \draw[muiM=cRed, line width=1.8pt] (h3) -- (dots);
    \draw[muiM=cRed, line width=1.8pt] (dots) -- node[above, font=\small, text=cRed] {$W_{hh}$} (hT);
    \node[ny, above=of hT] (yT) {$\hat y$};
    \draw[muiM=cOrange!70!black] (hT) -- node[right, font=\small] {many-to-one} (yT);
    \node[hop=cPurple, above=6mm of h2, font=\small] {mọi bước dùng chung $W_{xh}, W_{hh}, \mathbf{b}$};
  \end{scope}
  \node[below=5mm of xc, nhan] {dạng cuộn};
  \node[below=5mm of x2, hop=cBlue, font=\small] {$h_t = \tanh(W_{xh}x_t + W_{hh}h_{t-1} + \mathbf{b})$};
\end{tikzpicture}

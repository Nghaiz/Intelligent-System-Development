% Mặt nạ dropout (p ≈ 0,3) trên 8 đơn vị ẩn × 12 bước: dropout thường đổi mặt nạ ở mỗi bước,
% recurrent dropout dùng một mặt nạ cho cả chuỗi. Mặt nạ sinh giả ngẫu nhiên tất định bằng pgfmath.
\begin{tikzpicture}[x=4.2mm, y=4.2mm, font=\footnotesize]
  \pgfmathsetseed{7}
  \foreach \t in {0,...,11} \foreach \h in {0,...,7} {
    \pgfmathsetmacro{\r}{rnd}
    \ifdim\r pt<0.3pt \def\c{accentRed!75}\else\def\c{lightBlue}\fi
    \fill[\c, draw=white] (\t,-\h) rectangle ++(1,1);
  }
  \node[above, align=center] at (6,1.2) {Dropout thường:\\mặt nạ mới ở mỗi bước $t$};
  \begin{scope}[xshift=7.2cm]
    \foreach \h/\drop in {0/0, 1/1, 2/0, 3/0, 4/1, 5/0, 6/0, 7/1} {
      \ifnum\drop=1 \def\c{accentRed!75}\else\def\c{lightBlue}\fi
      \foreach \t in {0,...,11} \fill[\c, draw=white] (\t,-\h) rectangle ++(1,1);
    }
    \node[above, align=center] at (6,1.2) {Recurrent dropout:\\một mặt nạ cho cả chuỗi};
  \end{scope}
  \foreach \x in {0, 17.15} {
    \draw[-{Stealth}] (\x,-7.6) -- ++(12.5,0) node[right] {$t$};
    \node[left] at (\x,0.5) {$h_1$};
    \node[left] at (\x,-6.5) {$h_8$};
  }
  \fill[accentRed!75] (11,-9.2) rectangle ++(0.8,0.8); \node[right] at (11.9,-8.8) {đơn vị bị tắt};
  \fill[lightBlue] (18,-9.2) rectangle ++(0.8,0.8); \node[right] at (18.9,-8.8) {đơn vị giữ lại};
\end{tikzpicture}

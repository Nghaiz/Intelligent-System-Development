% Mặt nạ dropout (p ≈ 0,3) trên 8 đơn vị ẩn × 12 bước: dropout thường đổi mặt nạ ở mỗi bước,
% recurrent dropout dùng một mặt nạ cho cả chuỗi. Mặt nạ trái sinh giả ngẫu nhiên tất định bằng pgfmath.
\begin{tikzpicture}[x=5.4mm, y=5.4mm, font=\small]
  \pgfmathsetseed{7}
  \foreach \t in {0,...,11} \foreach \h in {0,...,7} {
    \pgfmathsetmacro{\r}{rnd}
    \ifdim\r pt<0.3pt \def\c{cRed!80}\else\def\c{cBlue!35}\fi
    \fill[\c, draw=white, thick, rounded corners=1pt] (\t,-\h) rectangle ++(1,1);
  }
  \node[tieude, align=center] at (6,2.5) {Dropout thường:\\mặt nạ mới ở mỗi bước $t$};
  \begin{scope}[xshift=8.4cm]
    \foreach \h/\drop in {0/0, 1/1, 2/0, 3/0, 4/1, 5/0, 6/0, 7/1} {
      \ifnum\drop=1 \def\c{cRed!80}\else\def\c{cGreen!40}\fi
      \foreach \t in {0,...,11} \fill[\c, draw=white, thick, rounded corners=1pt] (\t,-\h) rectangle ++(1,1);
    }
    \node[tieude, align=center] at (6,2.5) {Recurrent dropout:\\một mặt nạ cho cả chuỗi};
  \end{scope}
  \foreach \x in {0, 15.55} {
    \draw[-{Stealth}, very thick, cNavy] (\x,-7.6) -- ++(12.6,0) node[right] {$t$};
    \node[left] at (\x,0.5) {$h_1$};
    \node[left] at (\x,-6.5) {$h_8$};
  }
  \fill[cRed!80, rounded corners=1pt] (8,-9.3) rectangle ++(0.9,0.9); \node[right] at (9.0,-8.85) {đơn vị bị tắt};
  \fill[cBlue!35, rounded corners=1pt] (14,-9.3) rectangle ++(0.9,0.9); \fill[cGreen!40, rounded corners=1pt] (15,-9.3) rectangle ++(0.9,0.9);
  \node[right] at (16.0,-8.85) {đơn vị giữ lại};
\end{tikzpicture}

% Khối Residual: lan truyền xuôi y = F(x) + x (xanh) và dòng gradient ngược qua kết nối tắt (đỏ).
\begin{tikzpicture}[node distance=6mm, font=\small,
  k/.style={hop=#1, minimum width=38mm, minimum height=9mm},
  tron/.style={circle, inner color=white, outer color=cOrange!60, draw=cOrange!60!black, very thick, font=\large\bfseries, inner sep=1pt}]
  \fill[cGreen!6, rounded corners=10pt, draw=cGreen!40, thick] (-2.6,0.6) rectangle (2.6,-7.2);
  \node[k=cSlate, minimum width=16mm] (x) at (0,0) {$x$};
  \node[k=cGreen, below=of x] (c1) {Conv 3×3 – BN – ReLU};
  \node[k=cGreen, below=of c1] (c2) {Conv 3×3 – BN};
  \node[tron, below=6mm of c2] (p) {$+$};
  \node[k=cBlue, below=6mm of p, minimum width=20mm] (y) {ReLU $\to y$};
  \draw[mui] (x) -- (c1); \draw[mui] (c1) -- (c2); \draw[mui] (c2) -- node[right, font=\small\bfseries] {$F(x)$} (p); \draw[mui] (p) -- (y);
  \draw[muiM=cOrange, line width=2.6pt] (x.east) -- ++(2.0,0) |- node[pos=0.25, right, align=left, font=\small, text=cOrange!60!black]
    {\textbf{kết nối tắt}\\đồng nhất $I$, hoặc\\conv $1{\times}1$ khi đổi\\số kênh} (p.east);
  \draw[-{Stealth[length=3mm]}, cRed, line width=2.6pt, dashed] (-2.15,-6.6) -- node[left=8pt, align=right, font=\small, text=cRed]
    {$\partial E/\partial y$ qua\\nhánh $F$:\\có thể nhỏ dần} (-2.15,-0.2);
  \node[hop=cRed, anchor=west, text width=66mm, align=left] at (5.4,-3.3) {%
    \textbf{Lan truyền ngược qua khối:}\\[3pt]
    $\displaystyle \frac{\partial E}{\partial x} = \frac{\partial E}{\partial y}\Big(\frac{\partial F}{\partial x} + I\Big)$\\[4pt]
    Thành phần $I$ là “đường cao tốc”: dù $\partial F/\partial x \to 0$, gradient vẫn về tới các tầng đầu với độ
    lớn của $\partial E/\partial y$. Nhờ vậy ResNet-18 huấn luyện được từ đầu mà không thoái hóa.};

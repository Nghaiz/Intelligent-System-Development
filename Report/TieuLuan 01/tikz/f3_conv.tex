% Tích chập 5x5 * 3x3 (bộ lọc cạnh dọc) -> 3x3, số đã kiểm bằng NumPy.
\begin{tikzpicture}[x=6.5mm, y=6.5mm]
  \def\X{{{1,0,2,1,0},{0,1,3,1,1},{2,1,0,0,2},{1,0,1,2,1},{0,2,1,0,1}}}
  \def\Y{{{-2,0,2},{-1,-1,0},{1,1,-2}}}
  \foreach \i in {0,...,4} \foreach \j in {0,...,4} {
    \pgfmathtruncatemacro{\v}{\X[\i][\j]}
    \node[o] at (\j, -\i) {\v};}
  \begin{scope}[on background layer]
    \fill[lightBlue] (0.5, 0.5) rectangle (3.5, -2.5);
  \end{scope}
  \draw[ptitblue, very thick] (0.5, 0.5) rectangle (3.5, -2.5);
  \node[above] at (2, 0.6) {$X$ (5×5)};
  \node at (5.4, -2) {\Large $\ast$};
  \foreach \i in {0,...,2} \foreach \j/\k in {0/1, 1/0, 2/-1}
    \node[o, fill=lightOrange] at (6.5+\j, -1-\i) {\k};
  \node[above] at (7.5, -0.4) {$K$ (3×3)};
  \node at (10, -2) {\Large $=$};
  \foreach \i in {0,...,2} \foreach \j in {0,...,2} {
    \pgfmathtruncatemacro{\v}{\Y[\i][\j]}
    \node[o] at (11+\j, -1-\i) {\v};}
  \node[o, fill=lightBlue, draw=ptitblue, very thick] at (12, -1) {0};
  \node[above] at (12, -0.4) {$Y$ (3×3)};
  \draw[-{Stealth}, ptitblue, thick] (3.5, 0) to[bend left=25] (11.7, -0.75);
  \node[font=\footnotesize, align=left, anchor=north west, text width=88mm] at (-0.5, -5) {%
    $Y_{0,1} = \sum_{u=0}^{2}\sum_{v=0}^{2} X_{u,\,1+v}\,K_{u,v}
      = (0 - 1) + (1 - 1) + (1 - 0) = 0$: tổng tích của vùng tô xanh với bộ lọc.
    Bộ lọc này cho giá trị lớn khi cột trái sáng hơn cột phải, tức là phát hiện \emph{cạnh dọc};
    trượt cửa sổ (bước $s = 1$, không đệm) qua mọi vị trí được bản đồ đặc trưng $3{\times}3$.};
\end{tikzpicture}

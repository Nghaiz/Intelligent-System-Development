% Tế bào LSTM: đường cộng của cell state ở trên, bốn cổng (f, i, g = c̃, o) ở dưới.
\begin{tikzpicture}[font=\footnotesize, x=1cm, y=1cm,
  gate/.style={draw=ptitblue, fill=lightOrange, minimum width=8mm, minimum height=6mm, rounded corners=1pt},
  op/.style={circle, draw=ptitblue, fill=white, inner sep=0pt, minimum size=5.5mm, font=\small},
  line/.style={thick, ptitblue},
  arr/.style={thick, ptitblue, -{Stealth[length=2mm]}}]
  \draw[rounded corners=6pt, fill=lightBlue!50, draw=ptitblue] (0,0) rectangle (8.8,4.4);
  % cell state
  \node[op] (mf) at (1.6,3.6) {$\times$};
  \node[op] (pl) at (4.4,3.6) {$+$};
  \draw[arr] (-1,3.6) node[left] {$c_{t-1}$} -- (mf);
  \draw[arr] (mf) -- (pl);
  \draw[arr] (pl) -- (9.8,3.6) node[right] {$c_t$};
  \node[anchor=south, mZero!80!black] at (4.4,4.45) {đường cộng: $c_t = f_t \odot c_{t-1} + i_t \odot \tilde c_t$};
  % cổng
  \node[gate] (gf) at (1.6,1.5) {$\sigma$};
  \node[gate] (gi) at (3.2,1.5) {$\sigma$};
  \node[gate] (gg) at (4.4,1.5) {$\tanh$};
  \node[gate] (go) at (6.2,1.5) {$\sigma$};
  \node[op] (mi) at (4.4,2.55) {$\times$};
  \node[op] (mo) at (7.4,1.5) {$\times$};
  \node[gate, fill=lightGreen] (tc) at (7.4,2.55) {$\tanh$};
  % đầu vào [h_{t-1}; x_t] và nhãn cổng (nằm dưới đường vào, không có dây nào đi qua)
  \draw[line] (-1,0.65) node[left] {$h_{t-1}$} -- (6.2,0.65);
  \draw[line] (0.8,-0.8) node[below] {$x_t$} -- (0.8,0.65);
  \foreach \g in {gf, gi, gg, go} \draw[arr] (\g |- 0,0.65) -- (\g);
  \foreach \g/\lab in {gf/{$f_t$ quên}, gi/{$i_t$ vào}, gg/{$\tilde c_t$}, go/{$o_t$ ra}}
    \node[nhan, anchor=north] at (\g |- 0,0.55) {\lab};
  % nối cổng
  \draw[arr] (gf) -- (mf);
  \draw[arr] (gi) |- (mi);
  \draw[arr] (gg) -- (mi);
  \draw[arr] (mi) -- (pl);
  \draw[arr] (7.4,3.6) -- (tc);
  \draw[arr] (tc) -- (mo);
  \draw[arr] (go) -- (mo);
  \draw[arr] (mo) -- (8.3,1.5) |- (9.8,0.65) node[right] {$h_t = o_t \odot \tanh(c_t)$};
\end{tikzpicture}

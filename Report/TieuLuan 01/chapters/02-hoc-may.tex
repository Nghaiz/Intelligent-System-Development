\chapter{Các Kỹ Thuật Học Máy Cơ Bản}
\label{ch:hoc-may}

\section{Cơ sở lý thuyết các thuật toán học máy cốt lõi}

Học máy có giám sát giải hai lớp bài toán: \emph{hồi quy} (dự báo một giá trị liên tục) và \emph{phân loại} (gán
nhãn rời rạc). Cho tập huấn luyện $\{(\mathbf{x}^{(i)}, y^{(i)})\}_{i=1}^{N}$ với $\mathbf{x} \in \mathbb{R}^{D}$, ta chọn
một họ hàm $f_\theta$ và tìm tham số $\theta$ cực tiểu hóa một hàm mất mát trung bình trên dữ liệu. Bốn họ thuật toán
dưới đây khác nhau ở họ hàm và cách tối ưu \cite{raschka2015,goodfellow2016}; chúng cũng là các mô hình được hiện thực
và đo đạc trong phần sau của chương.

\subsection{Hồi quy tuyến tính (Linear Regression)}

Mô hình giả định đầu ra là tổ hợp tuyến tính của đặc trưng:
\begin{equation}
  \hat y = f(\mathbf{x}) = \mathbf{w}^\top\mathbf{x} + b = \sum_{j=1}^{D} w_j x_j + b ,
\end{equation}
và cực tiểu hóa sai số toàn phương trung bình
\begin{equation}
  \mathcal{L}_{\mathrm{MSE}}(\mathbf{w}, b) = \frac{1}{2N}\sum_{i=1}^{N}\big(\hat y^{(i)} - y^{(i)}\big)^2
    = \frac{1}{2N}\lVert X\mathbf{w} + b\mathbf{1} - \mathbf{y}\rVert_2^2 .
\end{equation}
Về hình học, ta tìm đường (siêu phẳng) làm tổng bình phương các \emph{phần dư} $y - \hat y$ nhỏ nhất
(Hình~\ref{fig:linreg}a). Nghiệm đóng là phương trình chuẩn $\mathbf{w}^* = (X^\top X)^{-1}X^\top\mathbf{y}$. Khi các
đặc trưng tương quan mạnh (đa cộng tuyến), $X^\top X$ gần suy biến và hệ số dao động lớn; \textbf{Ridge} thêm phạt
$\alpha\lVert\mathbf{w}\rVert_2^2$ để nghiệm thành $(X^\top X + \alpha I)^{-1}X^\top\mathbf{y}$, luôn khả nghịch. Tăng
$\alpha$ kéo mọi hệ số dần về 0 (Hình~\ref{fig:linreg}b), đổi một chút độ chệch lấy phương sai nhỏ hơn.

\hinhtikz{f2_linreg}{Hồi quy tuyến tính: (a) đường bình phương tối thiểu và phần dư; (b) đường co hệ số của Ridge theo
cường độ phạt $\alpha$ (minh họa).}{linreg}

\paragraph{Hạ gradient.} Khi $D$ hoặc $N$ lớn, thay vì nghịch đảo ma trận ta đi ngược hướng gradient của mất mát:
\begin{equation}
  \mathbf{w}^{(t+1)} = \mathbf{w}^{(t)} - \eta\,\nabla_{\mathbf{w}}\mathcal{L}\big(\mathbf{w}^{(t)}\big),
  \qquad \nabla_{\mathbf{w}}\mathcal{L}_{\mathrm{MSE}} = \tfrac{1}{N}X^\top(X\mathbf{w} - \mathbf{y}).
\end{equation}
Với hàm mất mát lồi như MSE hay entropy chéo của Logistic Regression, mặt mất mát có dạng một chiếc bát và mọi đường đi
đủ nhỏ đều về đáy (Hình~\ref{fig:gd3d}). Đây cũng là cơ chế huấn luyện mọi mạng nơ-ron ở các phần sau, chỉ khác là mặt
mất mát của mạng nhiều tầng không còn lồi.

\hinhtikz{f2_gd3d}{Hạ gradient trên mặt mất mát lồi $\mathcal{L}(\mathbf{w}) = 0{,}6w_1^2 + 1{,}1w_2^2 + 0{,}4w_1w_2$ với
tốc độ học $\eta = 0{,}3$; tọa độ các bước được tính bằng NumPy, đường nét đứt là hình chiếu xuống đáy.}{gd3d}

\subsection{Hồi quy Logistic (Logistic Regression)}

Với nhãn $y \in \{0, 1\}$, tổ hợp tuyến tính được ép về khoảng $(0, 1)$ bằng hàm sigmoid:
\begin{equation}
  p = P(y = 1 \mid \mathbf{x}) = \sigma(z) = \frac{1}{1 + e^{-z}}, \qquad z = \mathbf{w}^\top\mathbf{x} + b .
\end{equation}
Ước lượng hợp lý cực đại dẫn tới hàm mất mát entropy chéo nhị phân
\begin{equation}
  \mathcal{L}_{\mathrm{BCE}} = -\frac{1}{N}\sum_{i=1}^{N}\Big[y^{(i)}\log\hat y^{(i)} + (1 - y^{(i)})\log(1 - \hat y^{(i)})\Big],
  \label{eq:bce}
\end{equation}
phạt rất nặng các dự đoán tự tin nhưng sai (Hình~\ref{fig:sigmoid}b). Đạo hàm theo $z$ rút gọn thành
$\partial\mathcal{L}/\partial z = \hat y - y$, nên $\nabla_{\mathbf{w}}\mathcal{L} = \tfrac{1}{N}\sum_i (\hat y^{(i)} - y^{(i)})\mathbf{x}^{(i)}$.
Ranh giới quyết định $p = 0{,}5$ ứng với $z = 0$, tức là một siêu phẳng (Hình~\ref{fig:sigmoid}a). Với $K$ lớp, sigmoid
được thay bằng softmax $\hat y_k = e^{z_k}/\sum_j e^{z_j}$ và mất mát là entropy chéo $-\log \hat y_{y}$. Khi dữ liệu mất
cân bằng, \texttt{class\_weight="balanced"} nhân mất mát của mỗi mẫu với $N/(K\,N_k)$ để lớp hiếm có tiếng nói ngang lớp
phổ biến.

\hinhtikz{f2_sigmoid}{Hồi quy Logistic: (a) hàm sigmoid và ngưỡng 0,5; (b) mất mát entropy chéo theo xác suất dự
đoán.}{sigmoid}

\subsection{Cây quyết định và Rừng ngẫu nhiên (Random Forest)}

Cây quyết định chia không gian bằng chuỗi câu hỏi nhị phân “$x_j \le \theta$?”. Độ thuần của một nút $t$ có các lớp với tỉ
lệ $p_k$ được đo bằng chỉ số Gini hoặc entropy:
\begin{equation}
  \mathrm{Gini}(t) = 1 - \sum_{k=1}^{K} p_k^2, \qquad \mathrm{Entropy}(t) = -\sum_{k=1}^{K} p_k\log_2 p_k .
\end{equation}
Tại mỗi nút, thuật toán duyệt mọi cặp $(j, \theta)$ và chọn cặp làm giảm tạp chất nhiều nhất:
\begin{equation}
  \Delta\mathrm{Gini} = \mathrm{Gini}(t) - \Big(\frac{N_L}{N_t}\mathrm{Gini}(t_L) + \frac{N_R}{N_t}\mathrm{Gini}(t_R)\Big).
\end{equation}
Hình~\ref{fig:tree} là một cây độ sâu 2 \emph{huấn luyện thật} trên \SL{extra/ch2/tree/nodes/0/n} hồ sơ tiểu đường của
mục~\ref{sec:data2}. Câu hỏi đầu tiên mà thuật toán tự chọn là HbA1c, câu hỏi thứ hai là đường huyết, đúng hai đặc trưng
y khoa dùng để chẩn đoán. Chỉ với hai câu hỏi, cây đạt Accuracy \SL{extra/ch2/tree/accPct}\,\% và F1 lớp dương
\SL{extra/ch2/tree/f1} trên tập kiểm thử, một mốc tham chiếu quan trọng khi đánh giá các mô hình phức tạp hơn ở cuối
chương.

\hinhtikz{f2_tree}{Cây quyết định độ sâu 2 huấn luyện trên tập tiểu đường: ngưỡng, chỉ số Gini, số mẫu $N$ và tỉ lệ mắc
bệnh tại mỗi nút đều lấy từ mô hình đã khớp.}{tree}

Một cây sâu khớp rất sát dữ liệu huấn luyện nhưng có phương sai lớn: đổi vài mẫu là cây đổi hẳn. \textbf{Random Forest}
\cite{breiman2001} trung bình hóa $B$ cây với hai nguồn ngẫu nhiên (Hình~\ref{fig:forest}): mỗi cây học trên một mẫu
bootstrap (rút có hoàn lại $N$ mẫu, mỗi mẫu bootstrap chứa khoảng 63\,\% số điểm khác nhau) và mỗi lần chia chỉ xét một
tập con ngẫu nhiên $\sqrt{D}$ đặc trưng. Hai cơ chế này làm các cây ít tương quan với nhau; nếu $B$ cây có phương sai
$\sigma^2$ và hệ số tương quan $\rho$, phương sai của trung bình là $\rho\sigma^2 + \frac{1-\rho}{B}\sigma^2$, giảm mạnh
khi $\rho$ nhỏ, trong khi độ chệch gần như giữ nguyên.

\hinhtikz{f2_forest}{Random Forest: mỗi cây học trên một mẫu bootstrap với tập con đặc trưng ngẫu nhiên; kết quả là bỏ
phiếu đa số (phân loại) hoặc trung bình (hồi quy).}{forest}

\subsection{Máy vector hỗ trợ (Support Vector Machine)}

SVM \cite{cortes1995} tìm siêu phẳng phân tách có \emph{lề} lớn nhất tới các điểm gần nhất, cho phép vi phạm qua biến
nới lỏng $\xi_i$:
\begin{equation}
  \min_{\mathbf{w}, b, \boldsymbol{\xi}}\ \frac{1}{2}\lVert\mathbf{w}\rVert^2 + C\sum_{i=1}^{N}\xi_i
  \quad\text{với}\quad y^{(i)}(\mathbf{w}^\top\mathbf{x}^{(i)} + b) \ge 1 - \xi_i,\ \ \xi_i \ge 0 ,
\end{equation}
trong đó $C$ cân bằng giữa lề rộng và lỗi huấn luyện. Độ rộng lề bằng $2/\lVert\mathbf{w}\rVert$, và chỉ các điểm nằm
trên hoặc trong lề, gọi là \emph{vector hỗ trợ}, quyết định siêu phẳng (Hình~\ref{fig:svm}a). Bài toán đối ngẫu chỉ phụ
thuộc vào tích vô hướng giữa các mẫu, nên có thể thay bằng một hàm nhân. Nhân RBF
\begin{equation}
  K(\mathbf{x}, \mathbf{x}') = \exp\!\big(-\gamma\lVert\mathbf{x} - \mathbf{x}'\rVert^2\big)
\end{equation}
tương ứng với một ánh xạ lên không gian vô hạn chiều: dữ liệu hình vòng tròn không tách được bằng đường thẳng trên mặt
phẳng trở nên tách được bằng một mặt phẳng nằm ngang sau ánh xạ (Hình~\ref{fig:svm}b). Cái giá là bộ giải phải làm việc
với ma trận nhân $N \times N$: thời gian huấn luyện tăng cỡ $O(N^2)$ tới $O(N^3)$ và mô hình phải lưu toàn bộ vector hỗ
trợ. \textbf{SVR} là phiên bản hồi quy, phạt các điểm nằm ngoài ống sai số rộng $\varepsilon$. Với văn bản TF-IDF hàng
nghìn chiều, dữ liệu thường đã gần khả tách tuyến tính nên \textbf{LinearSVC} (nhân tuyến tính, độ phức tạp tuyến tính
theo $N$) là lựa chọn hợp lý hơn.

\hinhtikz{f2_svm}{Máy vector hỗ trợ: (a) siêu phẳng lề cực đại, các vector hỗ trợ được khoanh đỏ; (b) ý tưởng của nhân
RBF: điểm gần tâm được nâng cao, điểm xa ở thấp, một mặt cắt ngang tách hai lớp.}{svm}

\section{Khảo sát thực nghiệm và tiền xử lý ba bộ dữ liệu}

Mọi thí nghiệm trong tiểu luận đi theo cùng một quy trình (Hình~\ref{fig:pipeline}): khám phá dữ liệu, chia tập trước khi

\section{Mạng nơ-ron đa tầng (MLP) trên ba nền tảng: NumPy, Keras, PyTorch}

\subsection{Cơ sở lý thuyết và kiến trúc mạng}

Một MLP gồm các tầng kết nối đầy đủ. Tầng $l$ nhận kích hoạt $\mathbf{a}^{[l-1]}$ của tầng trước và tính
\begin{equation}
  \mathbf{z}^{[l]} = W^{[l]}\mathbf{a}^{[l-1]} + \mathbf{b}^{[l]}, \qquad \mathbf{a}^{[l]} = g^{[l]}\big(\mathbf{z}^{[l]}\big),
\end{equation}
với $g$ là ReLU ở tầng ẩn và sigmoid, softmax hoặc hàm đồng nhất ở tầng ra tùy bài toán. Lan truyền ngược áp dụng quy tắc
dây chuyền từ tầng ra về tầng vào:
\begin{align}
  \boldsymbol{\delta}^{[L]} &= \hat{\mathbf{y}} - \mathbf{y}, \\
  \boldsymbol{\delta}^{[l]} &= \big((W^{[l+1]})^\top\boldsymbol{\delta}^{[l+1]}\big) \odot g'^{[l]}\big(\mathbf{z}^{[l]}\big), \\
  \frac{\partial\mathcal{L}}{\partial W^{[l]}} &= \boldsymbol{\delta}^{[l]}\big(\mathbf{a}^{[l-1]}\big)^\top, \qquad
  \frac{\partial\mathcal{L}}{\partial\mathbf{b}^{[l]}} = \boldsymbol{\delta}^{[l]} .
\end{align}
Đẳng thức đầu đúng cho cả ba cặp đầu ra -- mất mát dùng ở đây (sigmoid + BCE, softmax + entropy chéo, tuyến tính + MSE
với hệ số 2), nên một hàm \texttt{backward} phục vụ cả ba bài toán. Trọng số khởi tạo theo He \cite{he2015init}
($\mathcal{N}(0, 2/n_{in})$) để biên độ kích hoạt ổn định qua các tầng ReLU. Bộ tối ưu Adam \cite{kingma2015} giữ trung bình
trượt của gradient $\mathbf{m}_t$ và của bình phương gradient $\mathbf{v}_t$, rồi cập nhật
\begin{equation}
  \mathbf{w} \leftarrow \mathbf{w} - \eta\,\frac{\hat{\mathbf{m}}_t}{\sqrt{\hat{\mathbf{v}}_t} + \epsilon},
  \qquad \hat{\mathbf{m}}_t = \frac{\mathbf{m}_t}{1 - \beta_1^t},\ \hat{\mathbf{v}}_t = \frac{\mathbf{v}_t}{1 - \beta_2^t},
\end{equation}
tức mỗi tham số có tốc độ học riêng. Dropout \cite{srivastava2014} tắt ngẫu nhiên một tỉ lệ $p$ nơ-ron mỗi lượt huấn luyện
(các dấu × ở Hình~\ref{fig:mlp}), buộc mạng không dựa vào một nơ-ron cụ thể.

Để so sánh ba nền tảng một cách công bằng, cả ba cài đặt dùng \emph{cùng một kiến trúc} (Hình~\ref{fig:mlp}): hai tầng ẩn
128 và 64 nơ-ron, ReLU, Dropout 0,2; Adam với tốc độ học $10^{-3}$, lô 256; dừng sớm khi mất mát kiểm định không giảm trong
8 epoch và khôi phục trọng số tốt nhất.

\paragraph{Thảo luận kết quả.}
\begin{itemize}
  \item \textbf{MLP tốt nhất trên dữ liệu y tế, nhưng chỉ ngang một cây hai tầng.} Cả ba bản MLP đạt F1 lớp dương quanh
    \SL{ch2_mlp_torch/diabetes/f1} (PyTorch), cao hơn Random Forest (\SL{ch2_sklearn/diabetes/rf/f1}) và vượt xa Logistic
    Regression (\SL{ch2_sklearn/diabetes/logreg/f1}). Tuy vậy cây quyết định độ sâu 2 ở Hình~\ref{fig:tree} cũng đạt F1
    \SL{extra/ch2/tree/f1}: phần lớn thông tin dự báo nằm trong hai ngưỡng HbA1c và đường huyết, và các mô hình phức tạp
    chỉ khai thác thêm rất ít. Ma trận nhầm lẫn (Hình~\ref{fig:cm-diab-2}) cho thấy MLP gần như không báo động giả (Precision
    \SL{ch2_mlp_torch/diabetes/precision}) nhưng vẫn bỏ sót khoảng một phần ba ca bệnh (Recall
    \SL{ch2_mlp_torch/diabetes/recall}).
  \item \textbf{Khả năng xếp hạng gần như ngang nhau; khác biệt nằm ở ngưỡng.} ROC-AUC của bốn mô hình phi tuyến chỉ chênh
    nhau trong khoảng \SL{extra/ch2/SVM RBF/auc} -- \SL{extra/ch2/MLP PyTorch/auc}, Average Precision từ
    \SL{extra/ch2/Logistic/ap} (Logistic) đến \SL{extra/ch2/MLP PyTorch/ap} (MLP) (Hình~\ref{fig:roc-diab}). Khác biệt lớn về
    F1 giữa Logistic/SVM và MLP phần nhiều đến từ \emph{điểm cắt} trên đường cong. Hình~\ref{fig:threshold-diab} cho thấy F1 của
    MLP đạt cực đại \SL{extra/ch2/mlp_best_f1} tại ngưỡng \SL{extra/ch2/mlp_best_thr}; trong sàng lọc y tế, bỏ sót đắt hơn báo
    động giả, nên hạ ngưỡng về 0,3 cho Recall \SL{extra/ch2/mlp_rec_at_03} với Precision vẫn \SL{extra/ch2/mlp_prec_at_03}.
  \item \textbf{Random Forest vẫn nhỉnh nhất trên giá nhà.} $R^2$ của Random Forest (\SL{ch2_sklearn/house/rf/r2}) cao hơn
    MLP PyTorch (\SL{ch2_mlp_torch/house/r2}) một chút, hai mô hình bám nhau trên đồ thị phân tán (Hình~\ref{fig:house-pred}).
    Cả hai chỉ giải thích khoảng 62\,\% phương sai. Phần dư của Random Forest đối xứng quanh 0 nhưng rộng
    (Hình~\ref{fig:house-resid}): \SL{extra/ch2/house_resid_within1Pct}\,\% dự báo sai lệch không quá 1 tỷ và
    \SL{extra/ch2/house_resid_within2Pct}\,\% không quá 2 tỷ. Phần còn lại nằm ở thông tin dữ liệu không có (vị trí chính xác
    trong quận, dự án, chất lượng xây dựng). Cây quyết định chia trực tiếp không gian theo từng quận/huyện, một lợi thế tự
    nhiên trên dữ liệu bảng có nhiều biến phân loại.
  \item \textbf{Văn bản: MLP chính xác nhất nhưng lớp trung tính là điểm mù chung.} MLP đạt Accuracy cao nhất
    (\SL{ch2_mlp_scratch/reviews/accPct}\,\% với bản NumPy), F1 macro ngang các mô hình tuyến tính. Cả ba bản MLP quá khớp
    rất nhanh: mất mát kiểm định chạm đáy ở epoch 2 -- 3 rồi tăng (Hình~\ref{fig:loss-mlp}), vì
    \SL{ch2_mlp_torch/reviews/params} tham số học trên chưa đến 17.000 văn bản. Ma trận nhầm lẫn (Hình~\ref{fig:cm-rev}) cho
    thấy lớp trung tính (3 sao) bị lẫn sang cả hai phía: một nhận xét 3 sao thường vừa khen vừa chê.
  \item \textbf{Ba nền tảng cho kết quả tương đương.} Với cùng kiến trúc và siêu tham số, chênh lệch F1/$R^2$ giữa NumPy,
    Keras và PyTorch nằm trong khoảng dao động do khởi tạo ngẫu nhiên khác nhau (Bảng~\ref{tab:ch2-fw}). Bản NumPy tự viết
    đạt độ chính xác ngang thư viện công nghiệp, xác nhận các công thức lan truyền ngược được cài đặt đúng. Khác biệt nằm ở
    chi phí kỹ thuật: PyTorch chạy GPU, Keras có tệp lưu lớn nhất do chứa cả trạng thái bộ tối ưu.
\end{itemize}

\section{Đóng gói triển khai và giao diện Web thực tế}

\subsection{Kiến trúc triển khai hệ thống}

Toàn bộ mô hình của ba chương được phục vụ bởi một ứng dụng Flask duy nhất gồm ba tầng (Hình~\ref{fig:deploy}):
\begin{enumerate}
  \item \textbf{Tầng giao diện:} một trang HTML5/CSS3 với sáu thẻ, JavaScript gọi API bằng \texttt{fetch} và vẽ biểu đồ
    bằng SVG thuần, không phụ thuộc thư viện ngoài.
  \item \textbf{Tầng ứng dụng:} máy chủ Flask nhận yêu cầu \texttt{POST} JSON tại sáu endpoint
    \texttt{/api/diabetes}, \texttt{/api/house}, \texttt{/api/sentiment}, \texttt{/api/vision}, \texttt{/api/stock},
    \texttt{/api/churn}.
  \item \textbf{Tầng mô hình:} mọi mô hình (\texttt{.pt}, \texttt{.joblib}) được nạp lên RAM một lần lúc khởi động và suy
    luận trên CPU. Bộ tiền xử lý được dựng lại bằng chính hàm \texttt{load\_*} đã dùng khi huấn luyện nên dữ liệu người
    dùng đi qua đúng phép biến đổi mà mô hình đã học.
\end{enumerate}
